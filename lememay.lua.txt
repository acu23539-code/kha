local function savePos(file, pos)
    local data = {x = pos.X.Scale, xO = pos.X.Offset, y = pos.Y.Scale, yO = pos.Y.Offset}
    if writefile then writefile(file, HttpService:JSONEncode(data)) end
end

local function loadPos(file, default)
    if isfile and isfile(file) then
        local success, data = pcall(function() return HttpService:JSONDecode(readfile(file)) end)
        if success then return UDim2.new(data.x, data.xO, data.y, data.yO) end
    end
    return default
end

local function saveState(file, state)
    if writefile then writefile(file, HttpService:JSONEncode({enabled = state})) end
end

local function loadState(file, default)
    if isfile and isfile(file) then
        local success, data = pcall(function() return HttpService:JSONDecode(readfile(file)) end)
        if success then return data.enabled end
    end
    return default
end

local function InstantReset()
    local char = Player.Character
    if not char then return end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChild("Humanoid")
    if hrp and hum then
        hum.Health = 0
        hum:ChangeState(Enum.HumanoidStateType.Dead)
        task.delay(0.01, function()
            pcall(function() Player:LoadCharacter() end)
        end)
    end
end

local function getWeapon()
    local char = Player.Character
    if not char then return nil end
    return Player.Backpack:FindFirstChild("Bat") or char:FindFirstChild("Bat")
end

local function findSentry()
    local char = Player.Character
    if not char or not char:FindFirstChild("HumanoidRootPart") then return end
    local rootPos = char.HumanoidRootPart.Position
    for _, obj in pairs(workspace:GetChildren()) do
        if obj.Name:find("Sentry") and not obj.Name:lower():find("bullet") then
            local ownerId = obj.Name:match("Sentry_(%d+)")
            if ownerId and tonumber(ownerId) == Player.UserId then continue end
            local part = obj:IsA("BasePart") and obj or obj:IsA("Model") and (obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart"))
            if part and (rootPos - part.Position).Magnitude <= DETECTION_DISTANCE then
                return obj
            end
        end
    end
end

local function moveSentry(obj)
    local char = Player.Character
    if not char or not char:FindFirstChild("HumanoidRootPart") then return end
    for _, part in pairs(obj:GetDescendants()) do
        if part:IsA("BasePart") then part.CanCollide = false end
    end
    local root = char.HumanoidRootPart
    local cf = root.CFrame * CFrame.new(0, 0, PULL_DISTANCE)
    if obj:IsA("BasePart") then
        obj.CFrame = cf
    elseif obj:IsA("Model") then
        local main = obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart")
        if main then main.CFrame = cf end
    end
end

local function attackSentry()
    local char = Player.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    local weapon = getWeapon()
    if not hum or not weapon then return end
    if weapon.Parent == Player.Backpack then hum:EquipTool(weapon) end
    pcall(function() weapon:Activate() end)
    for _, r in pairs(weapon:GetDescendants()) do
        if r:IsA("RemoteEvent") then pcall(function() r:FireServer() end) end
    end
end

local function createBaseESP(plot, mainPart)
    if baseEspInstances[plot.Name] then baseEspInstances[plot.Name]:Destroy() end
    local billboard = Instance.new("BillboardGui")
    billboard.Name = "rznnq" .. plot.Name
    billboard.Size = UDim2.new(0, 150, 0, 75)
    billboard.StudsOffset = Vector3.new(0, 8, 0)
    billboard.AlwaysOnTop = true
    billboard.Adornee = mainPart
    billboard.MaxDistance = 2000
    billboard.Parent = plot
    
    local label = Instance.new("TextLabel")
    label.Size = UDim2.new(1, 0, 1, 0)
    label.BackgroundTransparency = 1
    label.TextScaled = true
    label.Font = Enum.Font.Arcade
    label.TextColor3 = Color3.fromRGB(0, 255, 0)
    label.TextStrokeTransparency = 0
    label.TextStrokeColor3 = Color3.new(0, 0, 0)
    label.Parent = billboard
    
    baseEspInstances[plot.Name] = billboard
    return billboard
end

local function updateBaseESP()
    if not plotsFolder then return end
    for _, plot in ipairs(plotsFolder:GetChildren()) do
        local purchases = plot:FindFirstChild("Purchases")
        local plotBlock = purchases and purchases:FindFirstChild("PlotBlock")
        local mainPart = plotBlock and plotBlock:FindFirstChild("Main")
        local billboard = baseEspInstances[plot.Name]
        local timeLabel = mainPart and mainPart:FindFirstChild("BillboardGui") and mainPart.BillboardGui:FindFirstChild("RemainingTime")

        if timeLabel and mainPart then
            billboard = billboard or createBaseESP(plot, mainPart)
            local label = billboard:FindFirstChildWhichIsA("TextLabel")
            if label then label.Text = timeLabel.Text end
        elseif billboard then
            billboard:Destroy()
            baseEspInstances[plot.Name] = nil
        end
    end
end
RunService.RenderStepped:Connect(updateBaseESP)

local function LoadDesyncHybridUI()
    local screenGui = Instance.new("ScreenGui", CoreGui)
    screenGui.Name = "DesyncHybridGui"
    screenGui.ResetOnSpawn = false

    local function saveHybridPos(pos)
        local data = {
            X_Scale = pos.X.Scale, X_Offset = pos.X.Offset,
            Y_Scale = pos.Y.Scale, Y_Offset = pos.Y.Offset
        }
        if writefile then
            pcall(function() writefile(fileDesyncHybrid, HttpService:JSONEncode(data)) end)
        end
    end

    local function loadHybridPos()
        if isfile and isfile(fileDesyncHybrid) then
            local success, data = pcall(function() return HttpService:JSONDecode(readfile(fileDesyncHybrid)) end)
            if success then return UDim2.new(data.X_Scale, data.X_Offset, data.Y_Scale, data.Y_Offset) end
        end
        return UDim2.new(0.5, -75, 0.5, -25)
    end

    local desyncBtn = Instance.new("TextButton", screenGui)
    desyncBtn.Name = "DesyncButton"
    desyncBtn.Size = UDim2.new(0, 150, 0, 50)
    desyncBtn.Position = loadHybridPos()
    desyncBtn.BackgroundColor3 = Color3.fromRGB(0, 200, 100) -- COLOR VERDE
    desyncBtn.BackgroundTransparency = 0.1
    desyncBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
    desyncBtn.Text = "DESYNC"
    desyncBtn.Font = Enum.Font.GothamBold
    desyncBtn.TextSize = 18
    desyncBtn.AutoButtonColor = false

    local corner = Instance.new("UICorner", desyncBtn)
    corner.CornerRadius = UDim.new(0, 10)

    local stroke = Instance.new("UIStroke", desyncBtn)
    stroke.Color = Color3.fromRGB(255, 255, 255)
    stroke.Thickness = 1.5
    stroke.Transparency = 0.5

    local dragging, dragInput, dragStart, startPos

    desyncBtn.InputBegan:Connect(function(input)
        if (input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch) then
            dragging = true
            dragStart = input.Position
            startPos = desyncBtn.Position
            
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                    saveHybridPos(desyncBtn.Position)
                end
            end)
        end
    end)

    desyncBtn.InputChanged:Connect(function(input)
        if (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            dragInput = input
        end
    end)

    RunService.RenderStepped:Connect(function()
        if dragging and dragInput then
            local delta = dragInput.Position - dragStart
            desyncBtn.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)

    local function equipCarpet()
        local char = Player.Character
        if not char then return end
        local backpack = Player:FindFirstChild("Backpack")
        if backpack then
            for _, tool in ipairs(backpack:GetChildren()) do
                if tool.Name:lower():find("carpet") then
                    char:FindFirstChildOfClass("Humanoid"):EquipTool(tool)
                    break
                end
            end
        end
    end

    local function instaReset()
        local char = Player.Character
        if not char then return end
        local hrp = char:FindFirstChild("HumanoidRootPart")
        local hum = char:FindFirstChild("Humanoid")
        
        if hrp and hum then
            equipCarpet()
            task.wait()
            hrp.CFrame = deathCoords
            
            local connection
            connection = RunService.Heartbeat:Connect(function()
                if not char or not char.Parent or hum.Health <= 0 then
                    connection:Disconnect()
                    return
                end
                hrp.CFrame = deathCoords
            end)
        end
    end

    desyncBtn.MouseButton1Click:Connect(function()
        desyncBtn.BackgroundColor3 = Color3.fromRGB(0, 120, 60)
        desyncBtn.Text = "RESETTING..."
        
        pcall(function()
            if raknet then raknet.desync(true) end
        end)
        
        instaReset()
        
        task.wait(1.5)
        desyncBtn.BackgroundColor3 = Color3.fromRGB(0, 200, 100)
        desyncBtn.Text = "DESYNC"
    end)
end

    SliderTrack.Position = UDim2.new(0, 0, 0.7, 0)
    SliderTrack.BackgroundColor3 = Color3.fromRGB(0, 40, 20)
    Instance.new("UICorner", SliderTrack)
    local SliderFill = Instance.new("Frame", SliderTrack)
    SliderFill.Size = UDim2.new(sliderValue, 0, 1, 0)
    SliderFill.BackgroundColor3 = Color3.fromRGB(0, 255, 120)
    Instance.new("UICorner", SliderFill)
    local Handle = Instance.new("Frame", SliderTrack)
    Handle.Size = UDim2.new(0, 14, 0, 24)
    Handle.AnchorPoint = Vector2.new(0.5, 0.5)
    Handle.Position = UDim2.new(sliderValue, 0, 0.5, 0)
    Handle.BackgroundColor3 = Color3.fromRGB(0, 255, 120)
    Instance.new("UICorner", Handle).CornerRadius = UDim.new(0, 4)
    Instance.new("UIStroke", Handle).Color = Color3.new(1, 1, 1)

    local function ActualizarSlider(inputX)
        local rel = math.clamp((inputX - SliderTrack.AbsolutePosition.X) / SliderTrack.AbsoluteSize.X, 0, 1)
        sliderValue = rel
        SliderFill.Size = UDim2.new(rel, 0, 1, 0)
        Handle.Position = UDim2.new(rel, 0, 0.5, 0)
        PercentLabel.Text = "TRIGGER SENSITIVITY: " .. math.floor(rel * 100) .. "%"
    end

    SliderTrack.InputBegan:Connect(function(i) if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then draggingSlider = true BorderFrame.Draggable = false ActualizarSlider(i.Position.X) end end)
    UIS.InputChanged:Connect(function(i) if draggingSlider and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then ActualizarSlider(i.Position.X) end end)
    UIS.InputEnded:Connect(function(i) if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then draggingSlider = false BorderFrame.Draggable = true end end)

    ToggleBtn.MouseButton1Click:Connect(function()
        flashEnabled = not flashEnabled
        ToggleBtn.Text = flashEnabled and "AUTO FLASH: ON" or "AUTO FLASH: OFF"
        ToggleBtn.TextColor3 = flashEnabled and Color3.fromRGB(0, 255, 120) or Color3.fromRGB(150, 150, 150)
        TogStroke.Color = flashEnabled and Color3.fromRGB(0, 255, 120) or Color3.fromRGB(50, 50, 50)
    end)

    MinBtn.Activated:Connect(function()
        minimized = not minimized
        saveState(fileFlashMin, minimized)
        Content.Visible = not minimized
        MinBtn.Text = minimized and "+" or "×"
        local targetSize = minimized and UDim2.new(0, 320, 0, 32) or UDim2.new(0, 320, 0, 160)
        TweenService:Create(BorderFrame, TweenInfo.new(0.3, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), {Size = targetSize}):Play()
    end)

    ProximityPromptService.PromptButtonHoldBegan:Connect(function(prompt)
        if not flashEnabled then return end
        local startTime = os.clock()
        local conn
        conn = RunService.RenderStepped:Connect(function()
            if not prompt or not prompt.Parent then conn:Disconnect() return end
            if (os.clock() - startTime) / prompt.HoldDuration >= sliderValue then
                conn:Disconnect()
                local tool = Player.Character and Player.Character:FindFirstChildOfClass("Tool")
                if tool then tool:Activate() end
            end
        end)
        prompt.PromptButtonHoldEnded:Connect(function() if conn then conn:Disconnect() end end)
    end)
end

local function LoadBoosterUI()
    local BoosterGui = Instance.new("ScreenGui", CoreGui)
    BoosterGui.Name = "StickBoosterGui"
    local MainB = Instance.new("Frame", BoosterGui)
    MainB.Size = UDim2.new(0, 140, 0, 110)
    MainB.Position = UDim2.new(1, -153, 0.29, 0)
    MainB.BackgroundColor3 = Color3.fromRGB(10, 10, 10)
    Instance.new("UICorner", MainB)
    Instance.new("UIStroke", MainB).Color = Color3.fromRGB(0, 255, 120)
    local TitleB = Instance.new("TextLabel", MainB)
    TitleB.Size = UDim2.new(1, 0, 0, 25)
    TitleB.Text = "STICK BOOSTER"
    TitleB.TextColor3 = Color3.fromRGB(0, 255, 120)
    TitleB.Font = Enum.Font.GothamBold
    TitleB.BackgroundTransparency = 1
    local SpeedBox = Instance.new("TextBox", MainB)
    SpeedBox.Size = UDim2.new(1, -24, 0, 26)
    SpeedBox.Position = UDim2.new(0, 12, 0, 35)
    SpeedBox.BackgroundColor3 = Color3.fromRGB(30, 30, 30)
    SpeedBox.Text = tostring(SPEED)
    SpeedBox.TextColor3 = Color3.new(1, 1, 1)
    Instance.new("UICorner", SpeedBox)
    SpeedBox:GetPropertyChangedSignal("Text"):Connect(function() SPEED = tonumber(SpeedBox.Text) or SPEED end)
    local Toggle = Instance.new("Frame", MainB)
    Toggle.Size = UDim2.new(0, 50, 0, 22)
    Toggle.Position = UDim2.new(0.5, -25, 1, -30)
    Toggle.BackgroundColor3 = Color3.fromRGB(50, 50, 50)
    Instance.new("UICorner", Toggle).CornerRadius = UDim.new(1, 0)
    local Knob = Instance.new("Frame", Toggle)
    Knob.Size = UDim2.new(0, 18, 0, 18)
    Knob.Position = UDim2.new(0, 2, 0.5, -9)
    Knob.BackgroundColor3 = Color3.new(1, 1, 1)
    Instance.new("UICorner", Knob).CornerRadius = UDim.new(1, 0)

    Toggle.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            boosterEnabled = not boosterEnabled
            TweenService:Create(Knob, TweenInfo.new(0.2), {Position = boosterEnabled and UDim2.new(1, -20, 0.5, -9) or UDim2.new(0, 2, 0.5, -9)}):Play()
            Toggle.BackgroundColor3 = boosterEnabled and Color3.fromRGB(0, 200, 80) or Color3.fromRGB(50, 50, 50)
            if boosterEnabled then
                speedConn = RunService.Heartbeat:Connect(function()
                    if not boosterEnabled then return end
                    local char = Player.Character
                    local hum = char and char:FindFirstChildOfClass("Humanoid")
                    local hrp = char and char:FindFirstChild("HumanoidRootPart")
                    if hrp and hum and hum.MoveDirection.Magnitude > 0 then
                        local vel = hum.MoveDirection * SPEED
                        hrp.AssemblyLinearVelocity = Vector3.new(vel.X, hrp.AssemblyLinearVelocity.Y, vel.Z)
                    end
                end)
            else
                if speedConn then speedConn:Disconnect() speedConn = nil end
            end
        end
    end)
end
